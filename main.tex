% Stile "Definizione" (titolo in grassetto, testo in corsivo)
\newtheoremstyle{appunti}
  {6pt}       % Spazio sopra
  {6pt}       % Spazio sotto
  {\itshape}  % Testo in corsivo
  {}          % Nessun rientro
  {\bfseries} % Titolo in grassetto
  {.}         % Punto dopo il titolo
  {0.5em}     % Spazio dopo il titolo
  {}          % Vuoto

\theoremstyle{appunti}
\newtheorem{definizione}{Definizione}[section]

% Stile "Esempio" (titolo grande e a capo, testo normale)
\newtheoremstyle{esempiostyle}
  {8pt}                                            % Spazio sopra
  {8pt}                                            % Spazio sotto
  {\normalfont}                                    % Testo normale (non corsivo)
  {}                                               % Nessun rientro
  {\bfseries\itshape\Large}                        % Titolo in corsivo+grassetto
  {}                                               % Nessuna punteggiatura dopo il titolo
  {1em}                                            % Spazio dopo il titolo
  {\thmname{#1}\ \thmnumber{#2}\ \thmnote{(#3)}\\} % Va a capo dopo il titolo

\theoremstyle{esempiostyle}
\newtheorem{esempio}{Esempio}[section]

% SEZIONI E INDICE
% =========================================================
\usepackage{titlesec} % Gestione avanzata dei titoli
\usepackage{tocloft}  % Personalizzazione indice

% Stile titoli (uniforme, solo differenza di grandezza)
\titleformat{\section}
  {\LARGE\bfseries}               % H1: molto grande e in grassetto
  {\thesection.}{0.6em}{}

\titleformat{\subsection}
  {\Large\bfseries}               % H2: grande e in grassetto
  {\thesubsection.}{0.6em}{}

\titleformat{\subsubsection}
  {\large\bfseries}               % H3: più piccolo ma coerente
  {\thesubsubsection.}{0.6em}{}

% Spaziatura prima/dopo i titoli
\titlespacing*{\section}{0pt}{2.2ex plus .5ex}{1.2ex}
\titlespacing*{\subsection}{0pt}{1.8ex plus .4ex}{0.9ex}
\titlespacing*{\subsubsection}{0pt}{1.4ex plus .3ex}{0.7ex}

% Profondità di numerazione e indice
\setcounter{secnumdepth}{3}             % Numera fino a subsubsection
\setcounter{tocdepth}{3}                % Mostra fino a subsubsection nell’indice
\cftsetindents{subsubsection}{3em}{3em} % Rientro subsubsection nell’indice

% =========================================================
% paragraphsection (numerato)
% =========================================================
% Nota: aumentando tocdepth/secnumdepth a 4, abiliti numerazione e TOC anche per \paragraph.
% Il wrapper \paragraphsection rende comodo l'uso, ma NON aggiunge voci extra oltre il comportamento di \paragraph.
\titleformat{\paragraph}
  {\normalsize\bfseries}            % Stile del titolo
  {\theparagraph.}{0.6em}{}         % Numerazione + spaziatura

% Spaziatura prima / dopo
\titlespacing*{\paragraph}{0pt}{1.0ex plus .3ex}{0.7ex}

% L'inserimento nell’indice richiede l'abbassamento della profondità
\setcounter{tocdepth}{4}
\setcounter{secnumdepth}{4}

% Comando personalizzato per uso comodo
\newcommand{\paragraphsection}[1]{%
  \paragraph{#1}\mbox{}%
}
